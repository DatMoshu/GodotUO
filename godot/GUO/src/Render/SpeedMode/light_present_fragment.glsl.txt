#version 450
// GUO speed lighting: restore RealtimeLightTarget's hue-preserving transfer.
// One nearest texel; raw HDR accumulation is not modified by this pass.
layout(set = 0, binding = 0) uniform sampler2D light_source;
layout(location = 0) out vec4 light_presented;
void main()
{
    vec4 light = texelFetch(light_source, ivec2(gl_FragCoord.xy), 0);
    float brightest = max(light.r, max(light.g, light.b));
    if (brightest > 1.0) light.rgb /= brightest;
    light_presented = light;
}
