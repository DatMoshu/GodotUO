#version 450
// ADR-0048 approximate soften: one separable box kernel, class/cover edges stop it.
layout(set=0,binding=0) uniform sampler2D light_buffer;
layout(set=0,binding=1) uniform sampler2D surface_buffer;
layout(push_constant,std430) uniform Blur { vec4 screen_direction; ivec4 settings; } pc;
layout(location=0) out vec4 colour;
void main() {
    ivec2 pixel=ivec2(gl_FragCoord.xy), size=ivec2(pc.screen_direction.xy);
    ivec2 direction=ivec2(pc.screen_direction.zw);
    int centre=int(texelFetch(surface_buffer,pixel,0).w+0.5)&31;
    float weight=float(pc.settings.x+1);
    vec4 sum=texelFetch(light_buffer,pixel,0)*weight;
    for(int side=-1;side<=1;side+=2) {
        for(int step=1;step<=10;step++) {
            if(step>pc.settings.x) break;
            ivec2 at=clamp(pixel+direction*(side*step),ivec2(0),size-1);
            int kind=int(texelFetch(surface_buffer,at,0).w+0.5)&31;
            if(kind!=centre) break;
            float w=float(pc.settings.x+1-step);
            sum+=texelFetch(light_buffer,at,0)*w; weight+=w;
        }
    }
    colour=sum/max(weight,1.0);
}
